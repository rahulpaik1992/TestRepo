python3 - <<'PY'
from pathlib import Path
import re
 
path = Path("apps/api/src/azure-connect.ts")
text = path.read_text(encoding="utf-8")
 
old_pattern = re.compile(
r'consentUrl\.searchParams\.set\(\s*'
r'"scope",\s*'
r'\[\s*'
r'`api://\$\{clientId\}/\.default`,\s*'
r'"https://management\.azure\.com/"\s*\+\s*'
r'"user_impersonation",\s*'
r'\]\.join\(" "\),\s*'
r'\);',
re.MULTILINE,
)
 
replacement = '''consentUrl.searchParams.set(
"scope",
"https://management.azure.com/.default",
);'''
 
text, count = old_pattern.subn(replacement, text, count=1)
 
if count == 0:
# Support the compact form if formatting differs.
start = text.find('consentUrl.searchParams.set(\n "scope",')
end_marker = '\n );'
 
if start == -1:
raise SystemExit("Could not locate the consent scope block.")
 
end = text.find(end_marker, start)
 
if end == -1:
raise SystemExit("Could not locate the end of the consent scope block.")
 
end += len(end_marker)
 
text = (
text[:start]
+ replacement
+ text[end:]
)
 
path.write_text(text, encoding="utf-8")
print("Azure admin-consent scope corrected.")
PY
